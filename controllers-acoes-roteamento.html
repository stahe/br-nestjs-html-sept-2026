import { AppModule } from './app.module';

async function bootstrap() {

    Vamos comentar este código:

    • linha 1: [NestFactory] é a “fábrica” de aplicações [NestJS];
    • linha 2: o módulo raiz da aplicação, que veremos mais adiante;
    • linha 4: a função de inicialização é assíncrona ([async]): construir a aplicação e iniciar o servidor leva tempo, e aguardamos ([await]) a conclusão de cada etapa;
    • linha 6: constrói a aplicação a partir de seu módulo raiz. O [NestJS] descobre então todos os controllers e serviços da aplicação e cria suas instâncias;
    • linha 8: inicia o servidor HTTP na porta 3000;
    • linha 12: chama a função de inicialização.

    import { Controller, Get, Param, Post } from '@nestjs/common';

@Controller('first')
export class FirstController {
  // GET /first
  @Get()
  index(): string {
    return 'Action [index] of the [first] controller';
  }

  // GET /first/action01
  @Get('action01')
  action01(): string {
    return 'Action [action01] of the [first] controller, GET request';
  }

  // POST /first/action01: mesmo caminho, outro método HTTP
  @Post('action01')
  action01Post(): string {
    return 'Action [action01Post] of the [first] controller, POST request';
  }

  // GET /first/products/17: [:id] é um parâmetro de rota
  @Get('products/:id')
  product(@Param('id') id: string): string {
    return `Product no. ${id}`;
  }

  // GET /first/products/17/reviews/3: dois parâmetros de rota
  @Get('products/:id/reviews/:number')
  review(@Param() params: { id: string; number: string }): string {
    return `Review no. ${params.number} of product no. ${params.id}`;
  }

  // GET /first/files/...: [*] aceita qualquer final de caminho
  @Get('files/*')
  file(@Param('0') path: string): string {
    return `Requested file: ${path}`;
  }
}

    Vamos detalhar este código:

    • linha 3: [@Controller('first')] define o prefixo comum a todas as rotas do controller: suas ações tratam as rotas de URL que começam com [/first];
    • linhas 6–7: [@Get()] sem argumentos: a ação [index] trata [GET /first];
    • linhas 12–13: a ação [action01] processa [GET /first/action01]. A URL é a concatenação do prefixo do controller com o caminho da ação;
    • linhas 18–19: o mesmo caminho, mas para o método POST: [@Post] substitui [@Get]. Também existem [@Put], [@Delete], [@Patch], [@All]... O nome do método [TypeScript] ([action01Post]) é arbitrário: apenas os decoradores importam para o roteamento;
    • Linhas 24–25: [:id] é um parâmetro de rota: esse segmento da URL pode assumir qualquer valor. O decorador [@Param('id')] injeta seu valor no parâmetro [id] da ação. Esse valor é sempre uma string: para [/first/products/17], [id] é ['17'], e não o número 17;
    • linhas 30–31: dois parâmetros de rota. [@Param()] sem argumento injeta um objeto que contém todos eles: [{ id: '17', number: '3' }];
    • linhas 36–37: o caractere [*] aceita qualquer final de URL, inclusive [/]. A parte correspondente é o parâmetro chamado [0].

    import { Controller, Get, Header, HttpCode, NotFoundException } from '@nestjs/common';

@Controller('responses')
export class ResponsesController {
  // uma string: enviada como está, com o tipo [text/html]
  @Get('html')
  html(): string {
    return '<h1>Action [html]</h1>';
  }

  // a mesma string, mas declarada como texto simples
  @Get('text')
  @Header('Content-Type', 'text/plain; charset=utf-8')
  text(): string {
    return '<h1>Action [text]</h1>';
  }

  // um objeto: serializado em JSON, com o tipo [application/json]
  @Get('json')
  json(): object {
    return { lastName: 'Martin', firstName: 'Jean', age: 42 };
  }

  // um array: também serializado em JSON
  @Get('array')
  array(): string[] {
    return ['Mme', 'Mlle', 'Mr'];
  }

  // um código de status escolhido: 201 (Created) em vez de 200 (OK)
  @Get('created')
  @HttpCode(201)
  created(): object {
    return { message: 'resource created' };
  }

  // sem resposta: o código 204 (No Content) significa "nada a exibir"
  @Get('empty')
  @HttpCode(204)
  empty(): void {}

  // um método assíncrono: o NestJS espera a promise ser resolvida
  @Get('promise')
  async promise(): Promise<object> {
    const start = Date.now();
    await new Promise((done) => setTimeout(done, 500));
    return { wait: `${Date.now() - start} ms` };
  }

  // uma exceção: o NestJS a transforma em uma resposta de erro
  @Get('error')
  error(): string {
    throw new NotFoundException('This resource does not exist');
  }
}

    Vamos detalhar este código:

    • linhas 6–9: uma string é enviada como está, com o cabeçalho [Content-Type: text/html; charset=utf-8];
    • linhas 12–16: o decorador [@Header] define um cabeçalho da resposta. Declarada como texto simples ([text/plain]), a mesma string deixa de ser interpretada pelo navegador;
    • linhas 19–22: um objeto é serializado como JSON, com o cabeçalho [Content-Type: application/json];
    • linhas 25–28: um array também é serializado como JSON;
    • linhas 31–35: [@HttpCode(201)] altera o código de status da resposta (o padrão é 200 para um GET);
    • linhas 38–40: o código 204 (No Content) significa “sem conteúdo”: a ação não retorna nada;
    • linhas 43–48: uma ação pode ser assíncrona ([async]): nesse caso, ela retorna uma promessa ([Promise]), e o [NestJS] espera que ela seja resolvida antes de responder. É o caso de todas as ações que consultam um banco de dados;
    • Linhas 51–54: Uma ação pode lançar uma exceção. As exceções do [NestJS] ([NotFoundException], [BadRequestException], [ForbiddenException]...) carregam, cada uma, um código de status HTTP (aqui 404), que o [NestJS] usa em sua resposta.

    2.4. Exemplo 04: Redirecionamentos

    import { Controller, Get, Param, Redirect } from '@nestjs/common';

@Controller()
export class RedirectsController {
  // redirecionamento fixo: o endereço e o código são conhecidos de antemão
  @Get('old')
  @Redirect('/new', 301)
  oldPage(): void {}

  @Get('new')
  newPage(): string {
    return 'You are on the [new] page';
  }

  // redirecionamento calculado: o objeto retornado substitui o endereço de @Redirect()
  @Get('go/:destination')
  @Redirect()
  go(@Param('destination') destination: string): { url: string } {
    return { url: `/page/${destination}` };
  }

  @Get('page/:name')
  page(@Param('name') name: string): string {
    return `You are on the [${name}] page`;
  }

  // redirecionamento para outro site
  @Get('documentation')
  @Redirect('https://docs.nestjs.com')
  documentation(): void {}
}

    Vamos detalhar este código:

    • linhas 6–8: [@Redirect('/new', 301)]: um redirecionamento permanente. O código 301 (Moved Permanently) indica que a página mudou de endereço em caráter permanente; o código padrão é 302 (Found), um redirecionamento temporário;
    • linhas 16–20: um redirecionamento calculado: [@Redirect()] sem argumento, e a ação retorna um objeto [{ url: ... }] que define o endereço de destino;
    • linhas 28–30: é possível redirecionar para outro site.

    import { Controller, Get, Req, Res } from '@nestjs/common';
import { Request, Response } from 'express';

@Controller('express')
export class ExpressController {
  // a requisição Express: método, URL, cabeçalhos...
  @Get('request')
  request(@Req() req: Request): object {
    return {
      method: req.method,
      url: req.url,
      path: req.path,
      browser: req.get('user-agent'),
    };
  }

  // a resposta Express: a action envia ela mesma a resposta
  @Get('response')
  response(@Res() res: Response): void {
    res.status(200).type('text/plain').send('Response sent by [res.send]');
  }

  // redirecionamento com a resposta Express
  @Get('redirect')
  redirect(@Res() res: Response): void {
    res.redirect('/express/response');
  }

  // passthrough: usamos a resposta (aqui para um cabeçalho), mas o NestJS
  // ainda envia o valor retornado pela própria action
  @Get('passthrough')
  passthrough(@Res({ passthrough: true }) res: Response): string {
    res.setHeader('X-Example', '05-express');
    return 'Response sent by NestJS, with an added header';
  }

  // ERRO a não cometer: @Res() sem passthrough, e nada é enviado
  @Get('blocked')
  blocked(@Res() res: Response): string {
    return 'This string will never be sent: the browser keeps waiting...';
  }
}

    Vamos detalhar este código:

    • linha 2: os tipos [Request] e [Response] do [Express];
    • Linhas 7–15: [@Req()] injeta a requisição do [Express]: método HTTP, URL, caminho, cabeçalhos ([req.get(...)])... ;
    • linhas 18–21: [@Res()] injeta a resposta do [Express]. Atenção: Assim que uma ação usa [@Res()], o [NestJS] presume que é ela quem envia a resposta e ignora o que ela retorna. Aqui, a ação define o código de status e o tipo, e envia o corpo com [res.send];
    • linhas 24–27: um redirecionamento 302 feito pelo [Express] com [res.redirect(url)];
    • linhas 31–35: Com a opção [passthrough: true], a ação usa a resposta (aqui, para adicionar um cabeçalho), mas deixa que o [NestJS] envie o que ela retorna. É o melhor dos dois mundos;
    • linhas 38–41: o erro a evitar: [@Res()] sem [passthrough] e sem chamadas a [res.send], [res.redirect], [res.render].... A string retornada é ignorada, nenhuma resposta é enviada e o navegador espera indefinidamente.

    import { Injectable } from '@nestjs/common';

export interface Product {
  id: number;
  name: string;
  price: number;
}

@Injectable()
export class ProductsService {
  private readonly products: Product[] = [
    { id: 1, name: 'pen', price: 1.5 },
    { id: 2, name: 'notebook', price: 3.2 },
    { id: 3, name: 'eraser', price: 0.8 },
  ];

  findAll(): Product[] {
    return this.products;
  }

  findById(id: number): Product | undefined {
    return this.products.find((p) => p.id === id);
  }
}
    • Linhas 3–7: A interface [Product] descreve um produto;
    • linha 9: o decorador [@Injectable()] transforma a classe em um serviço (também chamado de provider): o [NestJS] cria uma única instância dele, que fornece (injeta) a todos que precisam dela;
    • linhas 11–15: os dados, aqui em memória;
    • linhas 17–23: os dois métodos do serviço.

    import { Controller, Get, NotFoundException, Param } from '@nestjs/common';
import { Product, ProductsService } from './products.service';

@Controller('products')
export class ProductsController {
  // injeção de dependência: o NestJS fornece a instância de ProductsService
  constructor(private readonly productsService: ProductsService) {}

  @Get()
  list(): Product[] {
    return this.productsService.findAll();
  }

  @Get(':id')
  oneProduct(@Param('id') id: string): Product {
    const product = this.productsService.findById(Number(id));
    if (!product) {
      throw new NotFoundException(`Product no. ${id} not found`);
    }
    return product;
  }
}
    • linha 7: injeção de dependência. O controller declara em seu construtor que precisa de um [ProductsService]; o [NestJS] fornece a ele a instância única do serviço. O controller nunca chama [new ProductsService()]. A sintaxe [private readonly productsService: ProductsService] do [TypeScript] declara ao mesmo tempo o parâmetro e o campo [this.productsService];
    • Linhas 9–12: A ação delega ao serviço;
    • linhas 14–21: [Number(id)] converte o parâmetro de rota (uma string) em número. Um produto inexistente dispara uma exceção 404.

    import { Controller, Get } from '@nestjs/common';
import { ProductsService } from '../products/products.service';

@Controller('clients')
export class ClientsController {
  // um serviço de OUTRO módulo, exportado por esse módulo
  constructor(private readonly productsService: ProductsService) {}

  @Get('cart')
  cart(): object {
    const products = this.productsService.findAll();
    const total = products.reduce((sum, p) => sum + p.price, 0);
    return { client: 'Jean Martin', items: products.length, total };
  }
}
    • Linha 7: [ClientsController] recebe por injeção um serviço de outro módulo;
    • linhas 9–14: ele usa esse serviço para calcular o total de um carrinho de compras.
